cmake_minimum_required(VERSION 3.21)
project(REAStudio VERSION 0.3.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(REA_VERSION "${PROJECT_VERSION}-alpha")
if(MSVC)
  # All sources, including tests, contain UTF-8 literals. Match official Qt DLL CRT.
  add_compile_options(/utf-8)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)

math(EXPR REA_POINTER_BITS "${CMAKE_SIZEOF_VOID_P} * 8")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/REA_BUILD_METADATA.txt"
  "Version=${REA_VERSION}\nCompiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\nSystem=${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_PROCESSOR}\nPointerBits=${REA_POINTER_BITS}\nQt=${Qt6_VERSION}\n")
set(ZYDIS_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)

set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)

add_subdirectory(third_party/zydis EXCLUDE_FROM_ALL)
add_library(rea_core src/elf.cpp src/binary.cpp src/address.cpp src/decoder.cpp src/disassembly.cpp src/analysis_control.cpp src/function_analysis.cpp src/analysis_backend.cpp src/process_memory.cpp src/ir.cpp src/control_flow_analysis.cpp src/control_flow_regions.cpp src/semantic_ir.cpp src/symbol_names.cpp)
target_link_libraries(rea_core PRIVATE Zydis PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(rea_core PRIVATE kernel32 dbghelp)
endif()
target_include_directories(rea_core PUBLIC include)
if(REA_HAS_CXA_DEMANGLE)
  target_compile_definitions(rea_core PRIVATE REA_HAS_CXA_DEMANGLE=1)
endif()
add_library(rea_ui src/window.cpp src/function_dialog.cpp include/rea/function_dialog.hpp src/process_view.cpp include/rea/process_view.hpp src/assembly_delegate.cpp src/theme.cpp src/program_models.cpp src/graph_view.cpp include/rea/graph_view.hpp src/instruction_model.cpp src/analysis_worker.cpp include/rea/analysis_worker.hpp)
set_target_properties(rea_ui PROPERTIES AUTOMOC ON)
target_link_libraries(rea_ui PUBLIC rea_core Qt6::Widgets)
add_executable(rea-studio src/main.cpp)
target_link_libraries(rea-studio PRIVATE rea_ui)
target_compile_definitions(rea_ui PRIVATE REA_VERSION="${REA_VERSION}")
target_compile_definitions(rea-studio PRIVATE REA_VERSION="${REA_VERSION}")
foreach(target rea_core rea_ui rea-studio)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
endforeach()

    message(FATAL_ERROR "REA_BUILD_LIBFUZZER requires Clang with libFuzzer")

    target_compile_definitions(rea_elf_fuzzer PRIVATE REA_HAS_CXA_DEMANGLE=1)

  target_link_options(rea_elf_fuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
